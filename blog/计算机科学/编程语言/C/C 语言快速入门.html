C 语言快速入门

计算机科学/编程语言/C/C 语言快速入门.md约 1,508 字

1. 环境准备

C 语言是编译型语言，需要通过编译器将源代码转换为可执行文件。常用的编译器有 GCC（GNU Compiler Collection）和 Clang。

检查编译器安装

# 检查 GCC 版本
gcc --version

# 检查 Clang 版本
clang --version

macOS 安装编译器

# 通过 Xcode Command Line Tools 安装（包含 Clang）
xcode-select --install

# 验证安装
clang --version
gcc --version   # macOS 上 gcc 通常是 clang 的别名

Linux 安装 GCC

# Ubuntu / Debian
sudo apt update
sudo apt install build-essential

# CentOS / RHEL
sudo yum groupinstall "Development Tools"

# 验证安装
gcc --version

Windows 安装

推荐使用 MinGW-w64 或 MSYS2，也可以使用 WSL (Windows Subsystem for Linux)。

2. 第一个程序

Hello, World!

创建文件 hello.c：

#include <stdio.h>

int main() {
    printf("Hello, World!\n");
    return 0;
}

编译与运行

# 编译（默认生成 a.out 或 a.exe）
gcc hello.c

# 运行
./a.out          # macOS / Linux
# 输出: Hello, World!

# 指定输出文件名
gcc hello.c -o hello
./hello

程序结构解析

  • #include <stdio.h> — 预处理指令，引入标准输入输出头文件，提供 printf 等函数
  • int main() — 程序入口函数，每个 C 程序有且仅有一个 main 函数
  • printf(...) — 格式化输出函数，\n 表示换行
  • return 0 — 函数返回值，0 表示程序正常退出

6. 运算符

算术运算符

int a = 10, b = 3;

a + b;    // 13  加法
a - b;    // 7   减法
a * b;    // 30  乘法
a / b;    // 3   整数除法（截断小数）
a % b;    // 1   取余（仅用于整数）

a++;      // 11  后置自增
++a;      // 12  前置自增
a--;      // 11  后置自减
--a;      // 10  前置自减

关系运算符

int a = 5, b = 3;

a == b;   // 0 (假)  等于
a != b;   // 1 (真)  不等于
a > b;    // 1       大于
a < b;    // 0       小于
a >= b;   // 1       大于等于
a <= b;   // 0       小于等于

C 语言中，0 表示假，非 0 表示真。

逻辑运算符

int a = 1, b = 0;

a && b;   // 0  逻辑与（两边都为真才为真）
a || b;   // 1  逻辑或（一边为真即为真）
!a;       // 0  逻辑非（取反）

短路求值：&& 左边为假时右边不执行；|| 左边为真时右边不执行。

位运算符

unsigned char a = 0b1100;  // 12
unsigned char b = 0b1010;  // 10

a & b;     // 0b1000 (8)   按位与
a | b;     // 0b1110 (14)  按位或
a ^ b;     // 0b0110 (6)   按位异或
~a;        // 按位取反
a << 1;    // 0b10000 (24) 左移
a >> 1;    // 0b0110 (6)   右移

赋值运算符

int a = 10;

a += 5;    // a = a + 5;    → 15
a -= 3;    // a = a - 3;    → 12
a *= 2;    // a = a * 2;    → 24
a /= 4;    // a = a / 4;    → 6
a %= 4;    // a = a % 4;    → 2
a &= 3;    // a = a & 3;
a |= 5;    // a = a | 5;
a <<= 1;   // a = a << 1;

三元运算符

// 条件 ? 表达式1 : 表达式2
// 条件为真，返回表达式1；否则返回表达式2

int a = 10, b = 20;
int max = (a > b) ? a : b;   // max = 20

常用格式符：

格式符 说明 示例
%d / %i 十进制整数 printf("%d", 42)

%s 字符串 printf("%s", "hello")

%o 八进制 printf("%o", 8)
%p 指针 printf("%p", &a)

scanf 格式化输入

#include <stdio.h>

int main() {
    int age;
    float height;
    char name[50];

    printf("请输入姓名: ");
    scanf("%s", name);        // 字符串不需要 &

    printf("请输入年龄: ");
    scanf("%d", &age);        // 基本类型需要 &（取地址）

    printf("请输入身高: ");
    scanf("%f", &height);

    printf("你好，%s！年龄 %d，身高 %.2f 米\n", name, age, height);

    return 0;
}

注意：scanf 读取字符串时遇到空格会停止。如需读取整行，使用 fgets。

8. 控制流程

if 条件语句

#include <stdio.h>

int main() {
    int score = 85;

    if (score >= 90) {
        printf("优秀\n");
    } else if (score >= 80) {
        printf("良好\n");
    } else if (score >= 60) {
        printf("及格\n");
    } else {
        printf("不及格\n");
    }

    return 0;
}

switch 多分支

#include <stdio.h>

int main() {
    int day = 3;

    switch (day) {
        case 1:
            printf("星期一\n");
            break;
        case 2:
            printf("星期二\n");
            break;
        case 3:
            printf("星期三\n");
            break;
        default:
            printf("其他\n");
    }

    return 0;
}

注意：每个 case 后必须加 break，否则会"穿透"到下一个 case。

for 循环

// 输出 1~10
for (int i = 1; i <= 10; i++) {
    printf("%d ", i);
}

// 从 10 到 1 倒序
for (int i = 10; i >= 1; i--) {
    printf("%d ", i);
}

// 嵌套循环：打印九九乘法表
for (int i = 1; i <= 9; i++) {
    for (int j = 1; j <= i; j++) {
        printf("%d*%d=%-2d ", j, i, i * j);
    }
    printf("\n");
}

while 循环

// 计算 1~100 的和
int sum = 0;
int i = 1;

while (i <= 100) {
    sum += i;
    i++;
}
printf("1+2+...+100 = %d\n", sum);

do-while 循环

// 至少执行一次
int i = 1;
do {
    printf("%d ", i);
    i++;
} while (i <= 10);

break 与 continue

// break：跳出整个循环
for (int i = 1; i <= 10; i++) {
    if (i == 5) {
        break;   // 到 5 就结束循环
    }
    printf("%d ", i);  // 输出 1 2 3 4
}

// continue：跳过当前迭代，继续下一次
for (int i = 1; i <= 10; i++) {
    if (i % 2 == 0) {
        continue;  // 跳过偶数
    }
    printf("%d ", i);  // 输出 1 3 5 7 9
}

goto 语句（慎用）

#include <stdio.h>

int main() {
    for (int i = 0; i < 10; i++) {
        for (int j = 0; j < 10; j++) {
            if (j == 5) {
                goto end_loop;   // 跳出多层循环
            }
        }
    }
end_loop:
    printf("循环结束\n");
    return 0;
}

9. 函数

函数定义与调用

#include <stdio.h>

// 函数声明（原型），也可以直接写在 main 前面

// 函数定义

函数参数

// 无参数、无返回值
void say_hello() {
    printf("Hello!\n");
}

// 有参数、有返回值
int max(int a, int b) {
    return (a > b) ? a : b;

指针基础

指针是一个变量，存储的是另一个变量的内存地址。

#include <stdio.h>

    int a = 10;

    printf("p 的值: %p\n", p);          // 0x...（和 &a 一样）
    printf("p 指向的值: %d\n", *p);     // 10（* 是解引用运算符）

    // 通过指针修改值
    *p = 20;
    printf("修改后 a 的值: %d\n", a);   // 20

    return 0;
}

void bubble_sort(int arr[], int len) {
    for (int i = 0; i < len - 1; i++) {
        for (int j = 0; j < len - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];

            }
        }
    }

void reverse_string(char *str) {
    int len = strlen(str);
    int left = 0, right = len - 1;
    while (left < right) {
        char temp = str[left];
        str[left] = str[right];
        str[right] = temp;
        left++;
        right--;
    }
}

    node->data = data;
    node->next = NULL;
    return node;
}

void print_list(Node *head) {

    while (p != NULL) {
        printf("%d -> ", p->data);
        p = p->next;
    }
    printf("NULL\n");
}

    head->next->next = create_node(3);
